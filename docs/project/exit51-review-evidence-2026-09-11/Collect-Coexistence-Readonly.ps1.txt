function Save($name,$value){
    $json=if($null -eq $value){'null'}elseif($value -is [array] -and $value.Count -eq 0){'[]'}else{ConvertTo-Json -InputObject $value -Depth 30}
    $bytes=[Text.UTF8Encoding]::new($false).GetBytes($json)
    $s=[IO.File]::Open((Join-Path $output $name),'CreateNew','Write','None')
    try{$s.Write($bytes,0,$bytes.Length);$s.Flush($true)}finally{$s.Dispose()}
}
function FileRecord([string]$path){
    try{$f=Get-Item -LiteralPath $path -ErrorAction Stop
        if($f.Attributes -band [IO.FileAttributes]::ReparsePoint){throw 'Reparse point refused.'}
        [pscustomobject]@{path=$path;bytes=$f.Length;sha256=(Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant();last_write_utc=$f.LastWriteTimeUtc.ToString('o');error=$null}
    }catch{[pscustomobject]@{path=$path;error=$_.Exception.Message}}
}
